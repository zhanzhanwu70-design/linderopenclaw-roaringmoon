1. selection-BeP8qtCb.js（事件產生層）

// handleToolExecutionStart - 工具開始執行
function handleToolExecutionStart(ctx, evt) {
  ctx.params.onAgentEvent?.({
    stream: "tool",
    data: { phase: "start", name: toolName, toolCallId, args }
  });
}

// handleToolExecutionUpdate - 工具執行更新
function handleToolExecutionUpdate(ctx, evt) {
  ctx.params.onAgentEvent?.({
    stream: "tool",
    data: { phase: "update", name: toolName, toolCallId, partialResult }
  });
}

// handleToolExecutionEnd - 工具執行結束
async function handleToolExecutionEnd(ctx, evt) {
  ctx.params.onAgentEvent?.({
    stream: "tool",
    data: { phase: "end", name: toolName, toolCallId, result }
  });
}

2. dispatch-acp-HFOhX_U-.js（事件投影層）

// emitToolSummary - 產生 tool summary 並發送
const emitToolSummary = async (event) => {
  if (!params.shouldSendToolSummaries) return;  // 🚫 檢查開關
  if (!isAcpTagVisible(settings, event.tag)) return;
  
  const renderedToolSummary = renderToolSummaryText(event);
  const toolSummary = truncateText(renderedToolSummary, ...);
  
  // 發送到頻道
  await params.deliver("tool", { text: toolSummary }, deliveryMeta);
};

3. dispatch-8E8vi2HV.js（Dispatch 層）

// createReplyDispatcher - 管理發送佇列
function createReplyDispatcher(options) {
  let sendChain = Promise.resolve();  // 串接 promise chain
  
  const enqueue = (kind, payload) => {
    sendChain = sendChain.then(async () => {
      await options.deliver(normalized, { kind });
    });
    return true;
  };
  
  return {
    sendToolResult: (payload) => enqueue("tool", payload),
    sendBlockReply: (payload) => enqueue("block", payload),
    sendFinalReply: (payload) => enqueue("final", payload),
  };
}

⚡ Realtime vs Final Only 模式

// deliveryMode 設定位置（dispatch-acp-HFOhX_U-.js） settings.deliveryMode === "live" → 即時發送，字元累積後立即 deliver settings.deliveryMode === "final_only" → 等待 done 事件才一次發出 // 當 deliveryMode === "final_only" 時的緩衝機制 const pendingToolDeliveries = []; // tool summary 先推到 pendingToolDeliveries pendingToolDeliveries.push({ payload: { text: toolSummary }, meta: deliveryMeta }); // 等 done 或 error 事件時才一口氣發送 await flush(true); // 強制 flush pendingToolDeliveries
